"use client";

/**
 * 05 차별점 4가지 — 카카오뱅크 인재영입 스크롤 텍스트 방식 (2026-10-15 사용자 요청)
 *  - 섹션은 스크롤 길이를 가진 래퍼(wrap) 안에서 화면 하나를 sticky 로 고정, 그 위에서 4문장이 순서대로 회색 → 시그니처 블루로 바뀐다.
 *  - 레이아웃: 연한 블루 배경 + 가운데 텍스트 + 좌우에 관련 사진 4장 콜라주.
 *  - 색 변화는 React 상태가 아니라 각 문장의 `--p`(0~1) CSS 변수를 직접 갱신 (리렌더·className 충돌 방지).
 *  - 괄호 ( ) 모션 버전은 `백업/차별점_괄호모션버전_2026-10-15/` 에 보관.
 */
import { useEffect, useRef } from "react";
import type { DifferentiatorsData } from "@/lib/sections";

const BG = "/media/4c4993e7-035e-490a-904c-3018cf878c27.webp";

export default function Differentiators({ title, data }: { title: string; data: DifferentiatorsData }) {
  const wrap = useRef<HTMLElement>(null);
  const lines = useRef<(HTMLLIElement | null)[]>([]);

  useEffect(() => {
    const el = wrap.current;
    if (!el) return;
    let raf = 0;
    const update = () => {
      raf = 0;
      const r = el.getBoundingClientRect();
      const total = r.height - window.innerHeight;
      const p = total > 0 ? Math.min(1, Math.max(0, -r.top / total)) : 1;
      const n = data.items.length;
      // 앞 12% 는 제목만 보이고, 이후 구간을 문장 수만큼 나눠 순서대로 채운다
      const q = Math.min(1, Math.max(0, (p - 0.12) / 0.8)) * n;
      lines.current.forEach((li, i) => li?.style.setProperty("--p", String(Math.min(1, Math.max(0, q - i)))));
    };
    const onScroll = () => { if (!raf) raf = requestAnimationFrame(update); };
    update();
    window.addEventListener("scroll", onScroll, { passive: true });
    window.addEventListener("resize", onScroll);
    return () => {
      window.removeEventListener("scroll", onScroll);
      window.removeEventListener("resize", onScroll);
      if (raf) cancelAnimationFrame(raf);
    };
  }, [data.items.length]);

  return (
    <section ref={wrap} className="diff-wrap" style={{ height: `${100 + data.items.length * 60}dvh` }}>
      <div className="diff-stick">
        <div className="diff-bg" aria-hidden />
        {/* 관련 이미지 4장 — 좌우로 나눠 살짝 기울여 배치. 모바일에서는 위·아래 2장만 */}
        <div className="diff-card diff-p1" aria-hidden style={{ backgroundImage: "url(/media/yt-hero.webp)" }} />
        <div className="diff-card diff-p2" aria-hidden style={{ backgroundImage: "url(/media/ig-hero.webp)" }} />
        <div className="diff-card diff-p3" aria-hidden style={{ backgroundImage: "url(/media/ba-face.webp)" }} />
        <div className="diff-card diff-p4" aria-hidden style={{ backgroundImage: `url(${BG})` }} />

        <div className="diff-body">
          <h2 className="diff-title">{title.replace(/\*\*/g, "")}</h2>
          <ol className="diff-list">
            {data.items.map((it, i) => (
              <li key={it.title} ref={(n) => { lines.current[i] = n; }} className="diff-line" style={{ ["--p" as string]: 0 }}>
                <svg className="diff-check" width="28" height="28" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="3" strokeLinecap="round" strokeLinejoin="round"><path d="M4 12.5l5 5L20 6.5" /></svg>
                <span>{it.title}</span>
              </li>
            ))}
          </ol>
        </div>
      </div>
    </section>
  );
}
